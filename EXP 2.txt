#include <REG51.h>

#define SEGPORT P2
sbit SS1 = P3^6;
sbit SS2 = P3^7;

const unsigned char LookupTable[] = {0x40, 0xF9, 0x24, 0x30, 0x19, 0x12, 0x02, 0x78, 0x00, 0x10};

void delaymS(unsigned long x)
{
    unsigned int i, j;
    for (i = 0; i < x; i++)
        for (j = 0; j < 1000; j++);
}

void main()
{
    int i = 0;
    unsigned char x = 1;
    SEGPORT = 0x00;
    SS1 = 1;
    SS2 = 1;
    while (1)
    {
        for (i = 0; i < 10; i++)
        {
            SEGPORT = ~LookupTable[i];
            delaymS(100);
        }
    }
}
